cmake_minimum_required(VERSION 3.16)

set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)

target_link_libraries(wasm_smoke PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
